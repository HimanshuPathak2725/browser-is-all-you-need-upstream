cmake_minimum_required(VERSION 3.16)
project(route-window-enumerator LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CHARM_IMPL "starter" CACHE STRING "starter, reference, negative, or mutation2")

if(CHARM_IMPL STREQUAL "reference")
  set(IMPL_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/.reference/route-window-enumerator.cpp")
  set(IMPL_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/.reference")
elseif(CHARM_IMPL STREQUAL "negative")
  set(IMPL_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/.meta/negative.cpp")
  set(IMPL_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/.meta/negative_include" "${CMAKE_CURRENT_SOURCE_DIR}")
elseif(CHARM_IMPL STREQUAL "mutation2")
  set(IMPL_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/.meta/mutations/semantic_2.cpp")
  set(IMPL_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/.meta/mutations/mutation2_include" "${CMAKE_CURRENT_SOURCE_DIR}")
else()
  set(IMPL_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/route-window-enumerator.cpp")
  set(IMPL_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}")
endif()

foreach(SUITE example hidden)
  if(SUITE STREQUAL "example")
    set(TEST_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/.meta/example.cpp")
  else()
    set(TEST_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/.meta/hidden_test.cpp")
  endif()
  add_executable(${SUITE} ${TEST_SOURCE} ${IMPL_SOURCE})
  target_include_directories(${SUITE} PRIVATE ${IMPL_INCLUDE})
  target_compile_options(${SUITE} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()

enable_testing()
add_test(NAME example COMMAND example)
add_test(NAME hidden COMMAND hidden)
